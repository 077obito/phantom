local function criarToggle(parent, nome, desc, callback)
    local yOffset = #parent:GetChildren() * 42
    
    local frame = Instance.new("Frame")
    frame.Size = UDim2.new(1, -10, 0, 36)
    frame.Position = UDim2.new(0, 5, 0, yOffset + 5)

    fCorner.Parent = frame
    
    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(1, -55, 1, 0)
    label.Position = UDim2.new(0, 10, 0, 0)
    label.BackgroundTransparency = 1
    label.Text = nome
    label.TextColor3 = Color3.fromRGB(220, 220, 240)
    label.TextSize = 14
    label.Font = Enum.Font.GothamSemibold
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Parent = frame
    
    local toggleBtnUI = Instance.new("ImageButton")
    toggleBtnUI.Size = UDim2.new(0, 40, 0, 22)
    toggleBtnUI.Position = UDim2.new(1, -48, 0.5, -11)
    toggleBtnUI.BackgroundColor3 = Color3.fromRGB(60, 60, 80)
    toggleBtnUI.BorderSizePixel = 0
    toggleBtnUI.Parent = frame
    
    local toggleCorner = Instance.new("UICorner")
    toggleCorner.CornerRadius = UDim.new(0, 11)
    toggleCorner.Parent = toggleBtnUI
    
    local toggleCircle = Instance.new("Frame")
    toggleCircle.Size = UDim2.new(0, 18, 0, 18)
    toggleCircle.Position = UDim2.new(0, 2, 0.5, -9)
    toggleCircle.BackgroundColor3 = Color3.fromRGB(200, 200, 200)
    toggleCircle.BorderSizePixel = 0
    toggleCircle.Parent = toggleBtnUI
    
    local circleCorner = Instance.new("UICorner")
    circleCorner.CornerRadius = UDim.new(0, 9)
    circleCorner.Parent = toggleCircle
    
    local active = false
    
    local function atualizarToggle()
        active = not active
        toggleBtnUI.BackgroundColor3 = active and Color3.fromRGB(100, 0, 255) or Color3.fromRGB(60, 60, 80)
        toggleCircle.Position = active and UDim2.new(0, 20, 0.5, -9) or UDim2.new(0, 2, 0.5, -9)
        toggleCircle.BackgroundColor3 = active and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(200, 200, 200)
        pcall(callback, active)
    end
    
    toggleBtnUI.MouseButton1Click:Connect(atualizarToggle)
    
    -- Atualizar canvas size
    parent.CanvasSize = UDim2.new(0, 0, 0, (#parent:GetChildren()) * 42 + 10)
    
    return { set = function(state)
        if state ~= active then
            atualizarToggle()
        end
    end, get = function() return active end }
end

-- LOOP PRINCIPAL (todas as funções rodam aqui)
-- ============================================================
runService.RenderStepped:Connect(function()
    local char = player.Character
    if not char then return end
    
    local hum = char:FindFirstChild("Humanoid")
    local root = char:FindFirstChild("HumanoidRootPart")
    if not hum or not root then return end
    
    -- WalkSpeed
    if flags.walkspeed then
        hum.WalkSpeed = 50
    elseif hum.WalkSpeed ~= 16 then
        hum.WalkSpeed = 16
    end
    
    -- JumpPower
    if flags.jumppower then
        hum.JumpPower = 100
    elseif hum.JumpPower ~= 50 then
        hum.JumpPower = 50
    end
    
    -- Noclip
    if flags.noclip then
        hum:ChangeState(11)
    end
    
    -- Fly
    if flags.fly then
        hum.PlatformStand = true
        local bv = root:FindFirstChild("PhantomBodyVelocity")
        local bg = root:FindFirstChild("PhantomBodyGyro")
        
        if not bv then
            bv = Instance.new("BodyVelocity")
            bv.Name = "PhantomBodyVelocity"
            bv.MaxForce = Vector3.new(10000, 10000, 10000)
            bv.Parent = root
        end
        if not bg then
            bg = Instance.new("BodyGyro")
            bg.Name = "PhantomBodyGyro"
            bg.MaxTorque = Vector3.new(10000, 10000, 10000)
            bg.Parent = root
        end
        
        local moveDir = Vector3.new()
        if userInput:IsKeyDown(Enum.KeyCode.W) then moveDir = moveDir + camera.CFrame.LookVector end
        if userInput:IsKeyDown(Enum.KeyCode.S) then moveDir = moveDir - camera.CFrame.LookVector end
        if userInput:IsKeyDown(Enum.KeyCode.A) then moveDir = moveDir - camera.CFrame.RightVector end
        if userInput:IsKeyDown(Enum.KeyCode.D) then moveDir = moveDir + camera.CFrame.RightVector end
        if userInput:IsKeyDown(Enum.KeyCode.Space) then moveDir = moveDir + Vector3.new(0, 1, 0) end
        if userInput:IsKeyDown(Enum.KeyCode.LeftShift) then moveDir = moveDir - Vector3.new(0, 1, 0) end
        
        if moveDir.Magnitude > 0 then
            moveDir = moveDir.Unit * flySpeed
        else
            moveDir = Vector3.new(0, 0, 0)
        end
        
        bv.Velocity = moveDir
        bg.CFrame = CFrame.new(root.Position, root.Position + camera.CFrame.LookVector)
    else
        hum.PlatformStand = false
        local bv = root:FindFirstChild("PhantomBodyVelocity")
        local bg = root:FindFirstChild("PhantomBodyGyro")
        if bv then bv:Destroy() end
        if bg then bg:Destroy() end
    end
    
    -- Invisivel
    if flags.invisivel then
        for _, part in pairs(char:GetDescendants()) do
            if part:IsA("BasePart") then
                part.Transparency = 1
            end
        end
    end
    
    -- ESP Players
    if flags.espPlayers then

MiscSection:NewButton("Server Hop", "Pula para outro servidor", function()
    local ts = game:GetService("TeleportService")
    local placeId = game.PlaceId
    local currentPlayer = game.Players.LocalPlayer

    -- Tenta um servidor diferente
    local function hop()
        local http = game:GetService("HttpService")
        local success, result = pcall(function()
            return http:JSONDecode(game:HttpGet("https://games.roblox.com/v1/games/" .. placeId .. "/servers/Public?limit=100"))
        end)
        if success then
            for _, server in pairs(result.data) do
                if server.playing < server.maxPlayers and server.id ~= game.JobId then
                    ts:TeleportToPlaceInstance(placeId, server.id, currentPlayer)
                    return
                end
            end
        end
    end
    hop()
end)
